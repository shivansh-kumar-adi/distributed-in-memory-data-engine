cmake_minimum_required(VERSION 3.20)
project(DistributedInMemoryDataEngine VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(AVENZA_BUILD_TESTS "Build tests" ON)
option(AVENZA_FETCH_GTEST "Download GoogleTest when not installed" OFF)
add_library(cacheforge
 src/cache/cache.cpp src/protocol/protocol.cpp src/persistence/persistence.cpp
 src/distributed/consistent_hash.cpp src/network/network.cpp)
target_include_directories(cacheforge PUBLIC include)
target_compile_features(cacheforge PUBLIC cxx_std_20)
if(WIN32)
 target_link_libraries(cacheforge PUBLIC ws2_32)
endif()
if(MSVC)
 target_compile_options(cacheforge PRIVATE /W4 /permissive-)
else()
 target_compile_options(cacheforge PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(avenza-server src/main.cpp)
target_link_libraries(avenza-server PRIVATE cacheforge)
add_executable(avenza-cli client/main.cpp)
target_link_libraries(avenza-cli PRIVATE cacheforge)
add_executable(avenza-cluster client/cluster_main.cpp)
target_link_libraries(avenza-cluster PRIVATE cacheforge)
add_executable(avenza-benchmark benchmarks/benchmark.cpp)
target_link_libraries(avenza-benchmark PRIVATE cacheforge)
if(AVENZA_BUILD_TESTS)
 enable_testing()
 add_executable(avenza-self-test tests/self_test.cpp)
 target_link_libraries(avenza-self-test PRIVATE cacheforge)
 add_test(NAME avenza-self-test COMMAND avenza-self-test)
 find_package(GTest CONFIG QUIET)
 if(NOT GTest_FOUND AND AVENZA_FETCH_GTEST)
  include(FetchContent)
  FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.zip DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(googletest)
 endif()
 if(TARGET GTest::gtest_main)
  add_executable(avenza-gtests tests/gtest_tests.cpp)
  target_link_libraries(avenza-gtests PRIVATE cacheforge GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(avenza-gtests)
 else()
  message(STATUS "GoogleTest unavailable; offline self-test remains enabled. Install GTest or set AVENZA_FETCH_GTEST=ON.")
 endif()
endif()
install(TARGETS avenza-server avenza-cli avenza-cluster avenza-benchmark RUNTIME DESTINATION bin)
